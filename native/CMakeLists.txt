cmake_minimum_required(VERSION 3.20)
project(ei_matlab_bridge LANGUAGES C CXX)
if(NOT EI_BINARY_NAME)
 set(EI_BINARY_NAME ei_inference)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJECT_PATH_MAX 240)
if(NOT EXISTS "${EI_MODEL_DIR}/model-parameters/model_metadata.h")
 message(FATAL_ERROR "EI_MODEL_DIR must be the inspected and patched build copy of the export")
endif()
set(SDK "${EI_MODEL_DIR}/edge-impulse-sdk")
file(GLOB_RECURSE TF_SOURCES CONFIGURE_DEPENDS "${SDK}/tensorflow/*.cc" "${SDK}/tensorflow/*.c")
list(FILTER TF_SOURCES EXCLUDE REGEX "/(cmsis_nn|cmsis-nn|esp_nn|arc_mli|ethos_u|ceva|hexagon|xtensa)/")
file(GLOB_RECURSE DSP_SOURCES CONFIGURE_DEPENDS "${SDK}/dsp/*.cpp")
file(GLOB_RECURSE MODEL_SOURCES CONFIGURE_DEPENDS "${EI_MODEL_DIR}/tflite-model/*.cpp" "${EI_MODEL_DIR}/tflite-model/*.cc" "${EI_MODEL_DIR}/tflite-model/*.c")
add_library(ei_inference SHARED ei_wrapper.cpp ei_model_adapter.cpp ei_image_adapter.cpp
 ei_detection_selector.cpp ei_porting.cpp ${TF_SOURCES} ${DSP_SOURCES} ${MODEL_SOURCES})
target_include_directories(ei_inference PRIVATE "${EI_MODEL_DIR}" "${SDK}" "${SDK}/third_party" "${SDK}/third_party/flatbuffers/include" PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(ei_inference PRIVATE EI_WRAPPER_BUILD EI_CLASSIFIER_TFLITE_ENABLE_CMSIS_NN=0
 EI_CLASSIFIER_TFLITE_ENABLE_ESP_NN=0 EI_CLASSIFIER_USE_FULL_TFLITE=0 TF_LITE_DISABLE_X86_NEON=1
 EIDSP_USE_CMSIS_DSP=0 EIDSP_LOAD_CMSIS_DSP_SOURCES=0 SILENCE_EI_CLASSFIER_OBJECT_DETECTION_COUNT_WARNING=1
 NOMINMAX _CRT_SECURE_NO_WARNINGS)
set_target_properties(ei_inference PROPERTIES PREFIX "" OUTPUT_NAME "${EI_BINARY_NAME}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(conf DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set_target_properties(ei_inference PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${conf} "${CMAKE_BINARY_DIR}/bin" ARCHIVE_OUTPUT_DIRECTORY_${conf} "${CMAKE_BINARY_DIR}/lib")
endforeach()
if(MSVC)
 target_compile_options(ei_inference PRIVATE /std:c++20 /EHsc /bigobj /FI"${CMAKE_CURRENT_SOURCE_DIR}/ei_msvc_compat.h")
else()
 target_compile_options(ei_inference PRIVATE -O2 -ffunction-sections -fdata-sections)
 if(MINGW)
  target_link_options(ei_inference PRIVATE -static-libgcc -static-libstdc++ -Wl,--gc-sections)
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(native_unit_tests ../tests/native_tests/test_algorithms.cpp ei_detection_selector.cpp ei_image_adapter.cpp)
 target_include_directories(native_unit_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME algorithms COMMAND native_unit_tests)
 add_executable(native_integration_tests ../tests/native_tests/test_inference.cpp)
 target_link_libraries(native_integration_tests PRIVATE ei_inference)
 set_target_properties(native_integration_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 foreach(conf DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(native_integration_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${conf} "${CMAKE_BINARY_DIR}/bin")
 endforeach()
 add_test(NAME actual_model COMMAND native_integration_tests)
 # Deterministic test-only backend: never used by the real inference target.
 add_library(ei_contract_fixture SHARED ei_wrapper.cpp ../tests/native_tests/fake_model_adapter.cpp)
 target_compile_definitions(ei_contract_fixture PRIVATE EI_WRAPPER_BUILD)
 target_include_directories(ei_contract_fixture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${EI_MODEL_DIR}")
 set_target_properties(ei_contract_fixture PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
 foreach(conf DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(ei_contract_fixture PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${conf} "${CMAKE_BINARY_DIR}/bin" ARCHIVE_OUTPUT_DIRECTORY_${conf} "${CMAKE_BINARY_DIR}/lib")
 endforeach()
 if(MINGW)
  target_link_options(ei_contract_fixture PRIVATE -static-libgcc -static-libstdc++)
 endif()
endif()
